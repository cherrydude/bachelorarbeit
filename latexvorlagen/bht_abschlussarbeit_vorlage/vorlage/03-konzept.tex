\begin{lstlisting}[caption=Summenfunktion.,label=lst:sum]
  # This function computes $\sum\limits_{i=1}^{n} i.$
  def sum(n):
    sum = 0
    for i in range(1,n+1):
        sum += i
    return sum 
\end{lstlisting}
